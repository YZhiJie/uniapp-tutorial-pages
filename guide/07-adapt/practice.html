7.3 实战案例：平台差异化能力与权限封装 ​

系统能力调用的难点从来不是 API 本身，而是授权流程、拒绝后的引导、平台能力差异三件事。本篇以定位与相机为例，产出生产可用的适配模块。

统一错误模型 ​

先定义业务层能识别的错误，页面不直接面对各平台原始报错：

ts
// src/platforms/permission/errors.ts
export class PermissionError extends Error {
  constructor(
    public code: 'DENIED' | 'RATIONALE' | 'UNSUPPORTED' | 'CANCELLED' | 'UNKNOWN',
    message: string
  ) {
    super(message)
    this.name = 'PermissionError'
  }
}

定位适配 ​

ts
// src/platforms/location/location.ts（公共逻辑）
export interface LngLat {
  longitude: number
  latitude: number
}

export function getCurrentLocation(): Promise<LngLat> {
  return new Promise((resolve, reject) => {
    uni.getLocation({
      type: 'gcj02',
      isHighAccuracy: false,
      highAccuracyExpireTime: 3000,
      success: (res) => resolve({ longitude: res.longitude, latitude: res.latitude }),
      fail: (err) => reject(mapLocationError(err))
    })
  })
}

function mapLocationError(err: UniApp.GetLocationFail): PermissionError {
  const msg = String(err.errMsg || '')
  if (msg.includes('auth deny') || msg.includes('authorize')) {
    return new PermissionError('DENIED', '定位权限被拒绝')
  }
  if (msg.includes('cancel')) {
    return new PermissionError('CANCELLED', '已取消')
  }
  return new PermissionError('UNKNOWN', msg)
}

H5 的浏览器定位是另一套授权 UI 与坐标系（WGS84），隔离在专属实现：

ts
// src/platforms/location/location.h5.ts
export function getCurrentLocation(): Promise<LngLat> {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) {
      reject(new PermissionError('UNSUPPORTED', '浏览器不支持定位'))
      return
    }
    navigator.geolocation.getCurrentPosition(
      (pos) =>
        resolve({
          longitude: pos.coords.longitude, // 实际业务需 WGS84→GCJ02 转换
          latitude: pos.coords.latitude
        }),
      (err) => {
        const map = {
          1: 'DENIED',
          2: 'UNSUPPORTED',
          3: 'UNKNOWN'
        } as const
        reject(new PermissionError(map[err.code] ?? 'UNKNOWN', err.message))
      },
      { enableHighAccuracy: true, timeout: 8000 }
    )
  })
}

授权申请与拒绝引导（小程序） ​

ts
// src/platforms/permission/permission.ts
type Scope = 'scope.userLocation' | 'scope.camera' | 'scope.album'

const SCOPE_LABEL: Record<Scope, string> = {
  'scope.userLocation': '位置',
  'scope.camera': '相机',
  'scope.album': '相册'
}

/** 确保已授权；拒绝过则弹引导跳设置页 */
export async function ensurePermission(scope: Scope): Promise<boolean> {
  // #ifdef MP-WEIXIN
  const setting = await new Promise<UniApp.GetSettingSuccess>((resolve) => {
    uni.getSetting({ success: resolve })
  })
  if (setting.authSetting[scope] === true) return true

  if (setting.authSetting[scope] === false) {
    // 曾经拒绝：无法再弹原生授权框，只能引导去设置页
    const { confirm } = await uni.showModal({
      title: '需要权限',
      content: `「${SCOPE_LABEL[scope]}」权限已被关闭，是否前往设置开启？`
    })
    if (!confirm) return false
    await uni.openSetting()
    return false // 返回后业务应重新触发一次流程
  }

  // 从未请求：发起授权
  const res = await new Promise<boolean>((resolve) => {
    uni.authorize({
      scope,
      success: () => resolve(true),
      fail: () => resolve(false)
    })
  })
  return res
  // #endif

  // #ifdef APP-PLUS
  // App 权限模型不同：android 权限在调用能力时触发，iOS 用 plus.navigator 检查
  // 此处交由系统能力调用时的 fail 回调兜底，直接放行尝试
  return true
  // #endif

  // #ifdef H5
  return true // H5 由具体能力（geolocation/getUserMedia）触发浏览器授权
  // #endif
}

拍照 / 相册统一入口 ​

ts
// src/platforms/media/pick-image.ts
export interface PickedImage {
  path: string
  size: number
}

export async function pickImage(source: 'camera' | 'album'): Promise<PickedImage> {
  const scope = source === 'camera' ? 'scope.camera' : 'scope.album'
  const granted = await ensurePermission(scope)
  if (!granted) throw new PermissionError('RATIONALE', '权限未开启')

  return new Promise((resolve, reject) => {
    uni.chooseImage({
      count: 1,
      sourceType: [source === 'camera' ? 'camera' : 'album'],
      sizeType: ['compressed'],
      success: (res) => {
        const file = res.tempFiles[0]
        resolve({ path: res.tempFilePaths[0], size: file.size })
      },
      fail: (err) =>
        reject(
          String(err.errMsg).includes('cancel')
            ? new PermissionError('CANCELLED', '已取消选择')
            : new PermissionError('UNKNOWN', String(err.errMsg))
        )
    })
  })
}

业务页面调用时只关心结果：

ts
import { pickImage } from '@/platforms/media/pick-image'
import { PermissionError } from '@/platforms/permission/errors'

async function onAvatarTap() {
  try {
    const img = await pickImage('camera')
    await uploadAvatar(img.path)
  } catch (e) {
    if (e instanceof PermissionError && e.code === 'CANCELLED') return
    uni.showToast({ title: (e as Error).message, icon: 'none' })
  }
}

manifest 权限声明（容易漏的一半） ​

运行时申请之外，安装期权限声明同样必须配齐：

  • Android：manifest.json → App 模块配置 / 权限，勾选定位、相机、存储；targetSdkVersion 30+ 需遵守分区存储。
  • iOS：配置 Privacy - Location When In Use Usage Description、相机/相册用途描述，描述不清会被 App Store 拒绝。
  • 微信小程序：app.json requiredPrivateInfos 声明 getLocation 等接口用途（后台类目也要匹配）。
  • H5：定位与摄像头仅在 HTTPS / localhost 可用。

权限适配验收清单 ​

  • [ ] 首次请求弹原生授权框，允许后功能正常
  • [ ] 拒绝后再次进入有明确引导，不出现点击无反应
  • [ ] 系统设置中关闭权限后，业务能捕获并提示
  • [ ] App 端 Android/iOS 各一台真机验证；注意 Android 13+ 的细粒度媒体权限
  • [ ] 小程序体验版（真机）验证，开发者工具的权限行为与真机不完全相同
  • [ ] H5 在 HTTP 环境给出「需 HTTPS」的优雅降级提示
