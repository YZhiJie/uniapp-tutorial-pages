6.3 实战案例：购物车状态模块设计 ​

购物车是状态管理的「教科书级」场景：跨页面共享、派生计算多、需要持久化、要与接口同步。本篇产出的 store 将在 模块 9 电商实战 中直接复用。

数据模型 ​

ts
// src/types/cart.ts
export interface CartSku {
  skuId: number
  goodsId: number
  title: string
  spec: string          // 规格描述，如“黑色 / XL”
  price: number         // 单价（单位：元）
  cover: string
  stock: number
  quantity: number
  selected: boolean
  invalid?: boolean     // 下架/无库存标记
}

Store 实现 ​

ts
// src/stores/cart.ts
import { computed, ref } from 'vue'
import { defineStore } from 'pinia'
import type { CartSku } from '@/types/cart'

export const useCartStore = defineStore(
  'cart',
  () => {
    const items = ref<CartSku[]>([])

    // ---- getters ----
    const totalCount = computed(() =>
      items.value.reduce((sum, item) => sum + item.quantity, 0)
    )
    const selectedItems = computed(() => items.value.filter((i) => i.selected && !i.invalid))
    const selectedCount = computed(() =>
      selectedItems.value.reduce((sum, item) => sum + item.quantity, 0)
    )
    const selectedAmount = computed(() =>
      Number(
        selectedItems.value
          .reduce((sum, item) => sum + item.price * item.quantity, 0)
          .toFixed(2)
      )
    )
    const allSelected = computed(
      () => items.value.length > 0 && items.value.every((i) => i.selected || i.invalid)
    )
    const tabBarBadge = computed(() => (totalCount.value > 99 ? '99+' : String(totalCount.value)))

    // ---- actions ----
    function add(payload: Omit<CartSku, 'quantity' | 'selected'>, quantity = 1) {
      const existed = items.value.find(
        (i) => i.skuId === payload.skuId && !i.invalid
      )
      if (existed) {
        existed.quantity = Math.min(existed.quantity + quantity, payload.stock)
      } else {
        items.value.unshift({ ...payload, quantity: Math.min(quantity, payload.stock), selected: true })
      }
      syncBadge()
    }

    function setQuantity(skuId: number, quantity: number) {
      const item = items.value.find((i) => i.skuId === skuId)
      if (!item) return
      item.quantity = Math.max(1, Math.min(quantity, item.stock))
    }

    function remove(skuId: number) {
      items.value = items.value.filter((i) => i.skuId !== skuId)
      syncBadge()
    }

    function removeBySkuIds(skuIds: number[]) {
      const set = new Set(skuIds)
      items.value = items.value.filter((i) => !set.has(i.skuId))
      syncBadge()
    }

    function toggleSelect(skuId: number) {
      const item = items.value.find((i) => i.skuId === skuId)
      if (item) item.selected = !item.selected
    }

    function toggleSelectAll(checked: boolean) {
      items.value.forEach((i) => {
        if (!i.invalid) i.selected = checked
      })
    }

    function clearSelected() {
      items.value = items.value.filter((i) => !i.selected)
      syncBadge()
    }

    function clear() {
      items.value = []
      syncBadge()
    }

    /** 后端最新商品信息回写（价格/库存/上下架） */
    function reconcile(latest: Array<Pick<CartSku, 'skuId' | 'price' | 'stock' | 'invalid'>>) {
      latest.forEach((info) => {
        const local = items.value.find((i) => i.skuId === info.skuId)
        if (!local) return
        local.price = info.price
        local.stock = info.stock
        local.invalid = info.invalid
        if (local.quantity > info.stock) local.quantity = info.stock
      })
    }

    function syncBadge() {
      if (totalCount.value === 0) {
        uni.removeTabBarBadge({ index: 1 }).catch(() => {})
      } else {
        uni.setTabBarBadge({ index: 1, text: tabBarBadge.value }).catch(() => {})
      }
    }

    return {
      items,
      totalCount,
      selectedItems,
      selectedCount,
      selectedAmount,
      allSelected,
      add,
      setQuantity,
      remove,
      removeBySkuIds,
      toggleSelect,
      toggleSelectAll,
      clearSelected,
      clear,
      reconcile,
      syncBadge
    }
  },
  {
    // 持久化插件（6.2 实现）：只存 items，金额等派生值不存
    persist: { key: 'cart:v1', paths: ['items'] }
  }
)

页面消费 ​

<!-- pages/cart/index.vue -->
<script setup lang="ts">
import { computed } from 'vue'
import { storeToRefs } from 'pinia'
import { onShow } from '@dcloudio/uni-app'
import { useCartStore } from '@/stores/cart'
import { cartApi } from '@/api/modules/cart'

const cartStore = useCartStore()
const { items, allSelected, selectedCount, selectedAmount } = storeToRefs(cartStore)

onShow(async () => {
  // 每次进入页面用服务端最新价格/库存校准本地购物车
  if (items.value.length) {
    const latest = await cartApi.snapshot(items.value.map((i) => i.skuId))
    cartStore.reconcile(latest)
  }
  cartStore.syncBadge()
})

}

在线演练场 ​

下面的演练用浏览器本地存储模拟了同样的 state/getter/action 结构与持久化，刷新页面后加购数量仍然保留：

设计要点回顾 ​

  1. 金额不存储，只计算：selectedAmount 是 computed，价格变动自动正确，杜绝“存出来的脏数据”。
  2. 库存是约束边界：所有修改数量的入口统一 clamp 到 [1, stock]，不在 UI 里散落判断。
  3. 失效商品隔离：invalid 商品不参与全选与结算，UI 置灰但保留记录，用户可手动删除。
  4. 本地优先 + 服务端校准：加购先写本地保证流畅，进购物车页 reconcile 最新价格库存；价格变化要在 UI 显著提示。
  5. 持久化最小集：只存 items，且带版本号 key，便于后续结构迁移。

练习 ​

  1. 给 store 增加「按店铺分组」：CartSku 增加 shopId，getter 输出分组结构与每组的小计。
  2. 实现购物车与服务端的增量同步：本地记录 dirty 标记，网络恢复后批量上报（提示：结合 uni.onNetworkStatusChange）。
  3. 为金额计算与 clamp 逻辑编写 Vitest 纯函数测试（把计算逻辑抽成不依赖 Pinia 的函数更易测）。
