6.2 进阶应用：Pinia 组合式 Store 与持久化 ​

setup store 的心智模型 ​

setup 写法对应概念
ref()state
computed()getter
普通函数action

return 出去的就是组件能访问的全部成员；不 return 的私有变量天然封装。

一个完整的 app store（含持久化） ​

ts
// src/stores/app.ts
import { ref } from 'vue'
import { defineStore } from 'pinia'

const THEME_KEY = 'app:theme'

type ThemeMode = 'light' | 'dark'

export const useAppStore = defineStore('app', () => {
  const theme = ref<ThemeMode>('light')
  const networkType = ref<'wifi' | '4g' | '5g' | 'none' | 'unknown'>('unknown')
  const systemInfo = ref<uni.getDeviceInfo.DeviceInfo | null>(null)

  function setTheme(mode: ThemeMode) {
    theme.value = mode
    uni.setStorageSync(THEME_KEY, mode)
  }

  function init() {
    const saved = uni.getStorageSync(THEME_KEY) as ThemeMode | ''
    if (saved === 'light' || saved === 'dark') theme.value = saved
    uni.getNetworkType({
      success: (res) => (networkType.value = res.networkType as typeof networkType.value)
    })
  }

  return { theme, networkType, systemInfo, setTheme, init }
})

App.vue onLaunch 调 appStore.init() 完成启动注水。

通用持久化：手写一个 Pinia 插件 ​

逐个字段手写 setStorage 容易漏，更通用的方式是 Pinia 插件：

ts
// src/stores/persist.ts
import type { PiniaPluginContext } from 'pinia'

export interface PersistOption {
  key: string
  /** 只持久化部分字段 */
  paths?: string[]
}

declare module 'pinia' {
  export interface DefineStoreOptionsBase<S> {
    persist?: PersistOption
  }
}

export function uniStoragePersist(context: PiniaPluginContext) {
  const { options, store } = context
  const persist = options.persist
  if (!persist) return

  // 启动注水
  const saved = uni.getStorageSync(persist.key)
  if (saved && typeof saved === 'object') {
    const patch = persist.paths
      ? Object.fromEntries(persist.paths.filter((k) => k in saved).map((k) => [k, saved[k]]))
      : saved
    store.$patch(patch)
  }

  // 变化后写回（$patch 合并，避免频繁同步写存储）
  let timer: ReturnType<typeof setTimeout> | null = null
  store.$subscribe(
    (_mutation, state) => {
      if (timer) clearTimeout(timer)
      timer = setTimeout(() => {
        const value = persist.paths
          ? Object.fromEntries(persist.paths.map((k) => [k, (state as Record<string, unknown>)[k]]))
          : state
        uni.setStorageSync(persist.key, value)
      }, 100)
    },
    { flush: 'post' }
  )
}

注册：

ts
// main.ts
import { createPinia } from 'pinia'
import { uniStoragePersist } from '@/stores/persist'

const pinia = createPinia()
pinia.use(uniStoragePersist)
app.use(pinia)

使用：

ts
export const useCartStore = defineStore('cart', () => {
  // ……
  return { items, totalCount, add, remove, clear }
}, {
  persist: { key: 'cart:v1', paths: ['items'] }
})

WARNING

  • setup store 默认不支持 $reset()，需要的话在 store 内实现一个 reset() action。
  • 小程序单 key 存储上限约 1MB，只持久化必要字段；列表类大缓存用分页 key 或 SQLite（11.2）。
  • 版本演进时给 key 带版本号（cart:v1），结构变更后按版本迁移而不是静默崩溃。

storeToRefs：解构不丢响应式 ​

ts
import { storeToRefs } from 'pinia'
const cartStore = useCartStore()
const { items, totalCount } = storeToRefs(cartStore) // 状态/ getter 用 storeToRefs
const { add, remove } = cartStore                    // action 直接解构

直接 const { items } = cartStore 会丢失响应式，是高频错误。

跨 store 组合：下单时联动用户与购物车 ​

ts
// src/stores/order.ts
import { useUserStore } from './user'
import { useCartStore } from './cart'
import { orderApi, type CreateOrderPayload } from '@/api/modules/order'

export const useOrderStore = defineStore('order', () => {
  const lastOrderId = ref(0)

  async function createOrder(payload: CreateOrderPayload) {
    const userStore = useUserStore()
    if (!userStore.isLoggedIn) throw new Error('UNAUTHORIZED')

    const res = await orderApi.create({ ...payload, userId: userStore.profile!.id })
    lastOrderId.value = res.id

    // 下单成功后清理已购买的购物车项
    const cartStore = useCartStore()
    cartStore.removeBySkuIds(payload.skuIds)
    return res
  }

  return { lastOrderId, createOrder }
})

在组件外使用 store ​

拦截器、请求封装等非组件环境中，确保在函数内部调用（此时 pinia 已安装）：

ts
// api/interceptors.ts
export function onRequestWithAuth(config: RequestConfig) {
  const token = useUserStore().token // ✅ 请求发生时调用
  // ……
}

不要在模块顶层 const userStore = useUserStore()，那时 pinia 可能尚未激活。

Vuex → Pinia 迁移对照 ​

Vuex 概念Pinia 对应迁移要点
stateref()直接声明，支持多 store
getterscomputed()无 rootGetters，直接 import 其他 store
mutations取消直接在 action 中修改状态
actions（含异步）普通函数（async）commit/dispatch 全部消失
namespaced module一个文件一个 store嵌套模块拍平成独立 store
mapState/mapMutationsstoreToRefs + 直接调Composition API 下不再需要映射辅助函数
严格模式默认就是安全的组件外也可改，但靠纪律/代码评审

Vuex 模块迁移示例：

ts
// 旧：store/modules/cart.js
// namespaced: true, state/mutations/addToCart/actions/syncCart

// 新：stores/cart.ts
export const useCartStore = defineStore('cart', () => {
  const items = ref<CartItem[]>([])
  function addToCart(goods: Goods) { items.value.push(/* … */) }
  async function syncCart() { /* await api … */ }
  return { items, addToCart, syncCart }
})

  // #ifdef H5 || MP-WEIXIN
