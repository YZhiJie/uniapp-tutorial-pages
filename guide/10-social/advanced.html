一、小程序静默登录 + 业务登录 ​

mermaid
sequenceDiagram
  participant C as 客户端
  participant WX as 微信
  participant S as 业务服务端
  C->>WX: uni.login() 获取 code
  C->>S: POST /auth/wx-login { code }
  S->>WX: code2session 换 openid/session_key
  S-->>C: token(短期) + refreshToken
  C->>C: 存储 token；需要资料时引导授权头像昵称
ts
// src/stores/user.ts
export const useUserStore = defineStore('user', () => {
  const token = ref(uni.getStorageSync('token') || '')
  const profile = ref<UserProfile | null>(null)

  const isLoggedIn = computed(() => Boolean(token.value))

  async function silentLogin() {
    // #ifdef MP-WEIXIN
    const { code } = await uni.login({ provider: 'weixin' })
    const res = await authApi.wxLogin(code)
    setSession(res.token, res.refreshToken)
    // #endif
  }

  async function loginByPassword(phone: string, password: string) {
    const res = await authApi.password({ phone, password })
    setSession(res.token, res.refreshToken)
  }

  function setSession(t: string, rt: string) {
    token.value = t
    uni.setStorageSync('token', t)
    uni.setStorageSync('refreshToken', rt)
  }

  function logout() {
    token.value = ''
    profile.value = null
    uni.removeStorageSync('token')
    uni.removeStorageSync('refreshToken')
  }

  return { token, profile, isLoggedIn, silentLogin, loginByPassword, logout }
})

App 启动时静默登录可以匿名态先跑（浏览内容），互动（点赞/评论/发消息）时再要求补全资料——这是提升转化的常见节奏。

二、Feed 流：游标分页 + 无限滚动 ​

ts
// src/api/modules/feed.ts
export const feedApi = {
  list(query: FeedQuery) {
    return http.get<FeedPage>('/feeds', { data: query as unknown as Record<string, unknown> })
  }
}
vue
<script setup lang="ts">
import { ref } from 'vue'
import { onReachBottom, onPullDownRefresh } from '@dcloudio/uni-app'

const list = ref<Feed[]>([])
const cursor = ref<string>()
const loading = ref(false)
const noMore = ref(false)

async function load(reset = false) {
  if (loading.value || (!reset && noMore.value)) return
  loading.value = true
  try {
    const res = await feedApi.list({ tab: 'recommend', size: 15, cursor: reset ? undefined : cursor.value })
    list.value = reset ? res.list : mergeFeed(list.value, res.list)
    cursor.value = res.nextCursor ?? undefined
    noMore.value = !res.nextCursor
  } finally {
    loading.value = false
  }
}

/** 按 id 去重合并（刷新可能把旧列表已有的内容带回来） */
function mergeFeed(oldList: Feed[], incoming: Feed[]): Feed[] {
  const map = new Map(oldList.map((f) => [f.id, f]))
  incoming.forEach((f) => map.set(f.id, f))
  return [...map.values()].sort((a, b) => Number(b.id) - Number(a.id))
}

onReachBottom(() => load())
onPullDownRefresh(async () => {
  await load(true)
  uni.stopPullDownRefresh()
})
load(true)
</script>

三、IM Socket 客户端（单例） ​

ts
// src/services/im/socket.ts
type Frame =
  | { type: 'message'; message: ChatMessageDTO }
  | { type: 'ack'; clientId: string; serverId: string }
  | { type: 'unread'; conversationId: string; unread: number }

type Listener = (frame: Frame) => void

class ImSocket {
  private task: UniApp.SocketTask | null = null
  private listeners = new Set<Listener>()
  private heartbeatTimer: ReturnType<typeof setInterval> | null = null
  private retry = 0
  private manualClosed = false

  connect(token: string) {
    this.manualClosed = false
    this.task = uni.connectSocket({
      url: `wss://im.example.com/ws?token=${encodeURIComponent(token)}`,
      complete: () => {}
    })
    this.task.onOpen(() => {
      this.retry = 0
      this.startHeartbeat()
    })
    this.task.onMessage((res) => {
      try {
        const frame = JSON.parse(String(res.data)) as Frame
        this.listeners.forEach((fn) => fn(frame))
      } catch {
        /* 忽略非法帧 */
      }
    })
    this.task.onClose(() => {
      this.stopHeartbeat()
      if (!this.manualClosed) this.scheduleReconnect(token)
    })
  }

  send(frame: unknown): Promise<void> {
    return new Promise((resolve, reject) => {
      this.task?.send({ data: JSON.stringify(frame), success: () => resolve(), fail: reject })
    })
  }

  subscribe(fn: Listener) {
    this.listeners.add(fn)
    return () => this.listeners.delete(fn)
  }

  close() {
    this.manualClosed = true
    this.task?.close({})
  }

  private startHeartbeat() {
    this.stopHeartbeat()
    this.heartbeatTimer = setInterval(() => {
      this.send({ type: 'ping' }).catch(() => {})
    }, 25000)
  }
  private stopHeartbeat() {
    if (this.heartbeatTimer) clearInterval(this.heartbeatTimer)
  }
  private scheduleReconnect(token: string) {
    const delay = Math.min(1000 * 2 ** this.retry++, 30000)
    setTimeout(() => this.connect(token), delay)
  }
}

export const imSocket = new ImSocket()

四、IM store：消息分发与会话重排 ​

ts
// src/stores/im.ts
export const useImStore = defineStore('im', () => {
  const conversations = ref<Conversation[]>([])
  const messages = reactive<Record<string, Message[]>>({})
  const totalUnread = ref(0)
  let unsubscribe: (() => boolean) | null = null

  function init() {
    if (unsubscribe) return
    unsubscribe = imSocket.subscribe((frame) => {
      if (frame.type === 'message') onIncoming(frame.message)
      else if (frame.type === 'ack') onAck(frame.clientId, frame.serverId)
      else if (frame.type === 'unread') updateUnread(frame.conversationId, frame.unread)
    })
  }

  function onIncoming(dto: ChatMessageDTO) {
    const msg = normalizeMessage(dto)
    pushMessage(msg)
    bumpConversation(msg)
    recomputeBadge()
  }

  function pushMessage(msg: Message) {
    const list = messages[msg.conversationId] ?? (messages[msg.conversationId] = [])
    if (!list.some((m) => m.id === msg.id)) list.push(msg)
  }

  function bumpConversation(msg: Message) {
    const conv = conversations.value.find((c) => c.id === msg.conversationId)
    if (conv) {
      conv.lastMessage = { type: msg.type, content: msg.content }
      conv.unread += 1
      conv.updatedAt = new Date(msg.createdAt).toISOString()
    }
    // 倒序排列
    conversations.value.sort((a, b) => +new Date(b.updatedAt) - +new Date(a.updatedAt))
  }

  async function sendText(conversationId: string, content: string) {
    const msg: Message = {
      id: cryptoUuid(),
      conversationId,
      fromId: useUserStore().profile!.id,
      type: 'text',
      content,
      createdAt: Date.now(),
      status: 'sending'
    }
    pushMessage(msg)
    bumpLocalConversation(msg)
    try {
      await imSocket.send({ type: 'message', clientId: msg.id, conversationId, content })
      msg.status = 'sent'
    } catch {
      msg.status = 'failed'
    }
  }

  function onAck(clientId: string, serverId: string) {
    for (const list of Object.values(messages)) {
      const target = list.find((m) => m.id === clientId)
      if (target) {
        target.serverId = serverId
        target.status = 'delivered'
        break
      }
    }
  }

  function recomputeBadge() {
    totalUnread.value = conversations.value
      .filter((c) => !c.muted)
      .reduce((sum, c) => sum + c.unread, 0)
    uni.setTabBarBadge({ index: 2, text: totalUnread.value > 99 ? '99+' : String(totalUnread.value) }).catch(() => {})
  }

  return { conversations, messages, totalUnread, init, sendText }
})

体验细节：

  • 键盘弹起时用 adjust-position 或自定义定位让输入框贴合键盘，务必真机验收。
  • 图片消息先显示缩略图 + 尺寸占位，点击 uni.previewImage。
  • 发送失败的消息点击重发，保持同一 clientId，避免产生两条。

练习 ​

  1. 实现消息去重与顺序保证：服务端重发/多端同步时如何按 serverId 与时间线合并？
  2. 完成「正在输入…」状态（提示：throttle 上报 typing 帧，对端超时清除）。
  3. 把消息列表落地到本地数据库（App SQLite 或 H5 IndexedDB），进入聊天页先渲染本地再同步。
