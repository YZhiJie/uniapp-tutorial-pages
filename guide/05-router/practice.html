5.3 实战案例：类型安全的路由表与跳转封装 ​

字符串 url 跳转有三大顽疾：路径写错运行时才炸、query 参数无类型提示、重构改路径要全局搜索。本篇用一张类型化路由表全部解决。

一、路由表：路径与参数形状集中声明 ​

ts
// src/router/routes.ts

/** 无参数页面 */
export interface RouteMap {
  '/pages/index/index': object
  '/pages/login/index': { redirect?: string }
  '/pages/goods/list': { category?: string; keyword?: string }
  '/pages/goods/detail': { id: number; from?: string }
  '/pages/order/confirm': { skuId: number; quantity: number }
  '/pages/order/list': { tab?: 'all' | 'unpaid' | 'done' }
  '/pages/address/picker': { preselect?: number }
  '/pages/cart/index': object
  '/pages/mine/index': object
}

export type RoutePath = keyof RouteMap
export type TabBarPath = '/pages/index/index' | '/pages/cart/index' | '/pages/mine/index'
export const TAB_BAR_PATHS: readonly string[] = [
  '/pages/index/index',
  '/pages/cart/index',
  '/pages/mine/index'
]

二、query 序列化 ​

ts
// src/router/query.ts
export function stringifyQuery(query?: Record<string, unknown>): string {
  if (!query) return ''
  const pairs = Object.entries(query)
    .filter(([, v]) => v !== undefined && v !== null && v !== '')
    .map(([k, v]) => `${encodeURIComponent(k)}=${encodeURIComponent(String(v))}`)
  return pairs.length ? `?${pairs.join('&')}` : ''
}

三、跳转封装：方法重载 + 鉴权 + 兜底 ​

ts
// src/router/index.ts
import { stringifyQuery } from './query'
import { TAB_BAR_PATHS, type RouteMap, type RoutePath } from './routes'

interface NavOptions {
  /** 需要登录；未登录时会重定向到登录页并带 redirect */
  auth?: boolean
  /** 失败时是否自动提示，默认 true */
  silent?: boolean
}

function buildUrl<P extends RoutePath>(path: P, query?: RouteMap[P]): string {
  return path + stringifyQuery(query as Record<string, unknown>)
}

function isTabPath(path: string): path is Extract<RoutePath, (typeof TAB_BAR_PATHS)[number]> {
  return TAB_BAR_PATHS.includes(path)
}

function callNav(
  api: 'navigateTo' | 'redirectTo' | 'reLaunch',
  url: string,
  silent?: boolean
): Promise<void> {
  return new Promise((resolve, reject) => {
    uni[api]({
      url,
      success: () => resolve(),
      fail: (err) => {
        if (!silent) {
          uni.showToast({ title: '页面跳转失败', icon: 'none' })
        }
        reject(err)
      }
    })
  })
}

/** 压栈跳转（保留当前页） */
export function navigateTo<P extends RoutePath>(
  path: P,
  query?: RouteMap[P],
  options: NavOptions = {}
) {
  return go('navigateTo', path, query, options)
}

/** 替换当前页 */
export function redirectTo<P extends RoutePath>(
  path: P,
  query?: RouteMap[P],
  options: NavOptions = {}
) {
  return go('redirectTo', path, query, options)
}

/** 清栈跳转 */
export function reLaunch<P extends RoutePath>(
  path: P,
  query?: RouteMap[P],
  options: NavOptions = {}
) {
  return go('reLaunch', path, query, options)
}

/** tabBar 页专用 */
export function switchTab(path: Extract<RoutePath, (typeof TAB_BAR_PATHS)[number]>) {
  return new Promise<void>((resolve, reject) => {
    uni.switchTab({
      url: path,
      success: () => resolve(),
      fail: reject
    })
  })
}

async function go<P extends RoutePath>(
  api: 'navigateTo' | 'redirectTo' | 'reLaunch',
  path: P,
  query: RouteMap[P] | undefined,
  options: NavOptions
): Promise<void> {
  // 鉴权内聚：调用方只需声明 auth: true
  if (options.auth && !uni.getStorageSync('token')) {
    const redirect = encodeURIComponent(buildUrl(path, query))
    return callNav('redirectTo', `/pages/login/index?redirect=${redirect}`, options.silent)
  }

  const url = buildUrl(path, query)
  if (isTabPath(path)) {
    return switchTab(path)
  }
  return callNav(api, url, options.silent)
}

export function navigateBack(delta = 1) {
  const pages = getCurrentPages()
  if (pages.length <= 1) {
    // 栈底无页可退（如扫码直达场景），回首页而不是无反应
    return switchTab('/pages/index/index')
  }
  uni.navigateBack({ delta })
}

四、目标页：类型化 query 解析 ​

ts
// src/router/use-query.ts
import { onLoad } from '@dcloudio/uni-app'
import { reactive } from 'vue'

export function useQuery<T extends Record<string, unknown>>(parser: {
  [K in keyof T]: (raw: string | undefined) => T[K]
}) {
  const query = reactive({}) as T

  onLoad((raw) => {
    for (const key in parser) {
      const value = raw?.[key]
      ;(query as Record<string, unknown>)[key] = parser[key](
        Array.isArray(value) ? value[0] : value === undefined ? undefined : String(value)
      )
    }
  })

  return query
}
ts
// pages/goods/detail.vue

  id: (v) => Number(v ?? 0),
  from: (v) => v ?? ''
})

六、与守卫的关系 ​

本篇封装把鉴权做成调用级选项（auth: true），比全局白名单更显式：

  • 全局守卫（5.2）负责兜底，防止遗漏。
  • 路由封装负责日常业务开发的类型安全与便捷调用。
  • 两者共用同一个 token 判断函数，避免规则分叉。

练习 ​

  1. 在路由表中新增「售后详情」页（id: number、orderId: number），故意漏传参数，观察 TS 报错；补全后跳转验证。
  2. 给 navigateTo 增加页面栈深度保护：接近 10 层时自动改用 redirectTo 并埋点上报。
  3. 实现 useQuery 对布尔与日期参数的解析器，并为路由表加上单元测试（参照本教程站点 Vitest 的做法）。
