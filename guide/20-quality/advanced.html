20.2 进阶应用：Store / Composable / 组件与契约测试 ​

纯逻辑测完，下一层是「有状态、有副作用」的代码。这一层的原则是：mock 边界，不 mock 实现——替换网络与平台能力，保留真实的 Pinia、响应式与组合逻辑。

一、测试 Pinia Store ​

以购物车 store 为例（setup 语法）：

ts
// src/stores/cart.ts
import { computed, ref } from 'vue'
import { defineStore } from 'pinia'
import { fetchCart, updateQuantity } from '@/api/cart'

export const useCartStore = defineStore('cart', () => {
  const lines = ref<Array<{ skuId: string; quantity: number }>>([])
  const count = computed(() => lines.value.reduce((sum, l) => sum + l.quantity, 0))

  async function load() {
    lines.value = await fetchCart()
  }

  async function setQuantity(skuId: string, quantity: number) {
    await updateQuantity(skuId, quantity)
    const line = lines.value.find((l) => l.skuId === skuId)
    if (line) line.quantity = quantity
  }

  return { lines, count, load, setQuantity }
})

测试只 mock API 模块，Pinia 本身用真实的：

ts
// tests/stores/cart.test.ts
import { beforeEach, describe, expect, it, vi } from 'vitest'
import { createPinia, setActivePinia } from 'pinia'
import { useCartStore } from '@/stores/cart'

vi.mock('@/api/cart', () => ({
  fetchCart: vi.fn(async () => [
    { skuId: 'sku-1', quantity: 2 },
    { skuId: 'sku-2', quantity: 1 }
  ]),
  updateQuantity: vi.fn(async () => undefined)
}))

beforeEach(() => {
  setActivePinia(createPinia())
  vi.clearAllMocks()
})

describe('cart store', () => {
  it('加载后派生总件数', async () => {
    const store = useCartStore()
    await store.load()
    expect(store.lines).toHaveLength(2)
    expect(store.count).toBe(3)
  })

  it('修改数量同时更新远端与本地', async () => {
    const store = useCartStore()
    await store.load()
    await store.setQuantity('sku-1', 5)
    expect(store.count).toBe(6)
    expect(store.lines.find((l) => l.skuId === 'sku-1')?.quantity).toBe(5)
  })

  it('远端失败时本地状态不变', async () => {
    const store = useCartStore()
    await store.load()
    const { updateQuantity } = await import('@/api/cart')
    vi.mocked(updateQuantity).mockRejectedValueOnce(new Error('network'))

    await expect(store.setQuantity('sku-1', 9)).rejects.toThrow('network')
    expect(store.lines.find((l) => l.skuId === 'sku-1')?.quantity).toBe(2)
  })
})

第三条用例是重点：失败路径不污染本地状态。只测 happy path 的 store 测试价值有限。

二、测试 Composable：时间与并发要可控 ​

带防抖的搜索 composable：

ts
// src/composables/useDebouncedSearch.ts
import { onUnload } from '@dcloudio/uni-app'
import { ref, watch } from 'vue'

export function useDebouncedSearch(
  keyword: () => string,
  search: (kw: string) => Promise<string[]>,
  delay = 300
) {
  const results = ref<string[]>([])
  const loading = ref(false)
  let timer: ReturnType<typeof setTimeout> | null = null

  watch(keyword, (kw) => {
    if (timer) clearTimeout(timer)
    timer = setTimeout(async () => {
      if (!kw.trim()) {
        results.value = []
        return
      }
      loading.value = true
      try {
        results.value = await search(kw)
      } finally {
        loading.value = false
      }
    }, delay)
  })

  onUnload(() => {
    if (timer) clearTimeout(timer)
  })

  return { results, loading }
}

测试用假定时器，避免真实等待：

ts
// tests/composables/useDebouncedSearch.test.ts
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest'
import { effectScope, nextTick, ref } from 'vue'
import { useDebouncedSearch } from '@/composables/useDebouncedSearch'

vi.mock('@dcloudio/uni-app', () => ({ onUnload: vi.fn() }))

describe('useDebouncedSearch', () => {
  beforeEach(() => vi.useFakeTimers())
  afterEach(() => vi.useRealTimers())

  it('300ms 内连续输入只触发一次搜索', async () => {
    const keyword = ref('')
    const search = vi.fn(async (kw: string) => [`result:${kw}`])
    const scope = effectScope()
    const { results } = scope.run(() => useDebouncedSearch(() => keyword.value, search))!

    keyword.value = 'a'
    await nextTick()
    keyword.value = 'ab'
    await nextTick()
    keyword.value = 'abc'
    await nextTick()

    expect(search).not.toHaveBeenCalled()
    await vi.advanceTimersByTimeAsync(300)
    expect(search).toHaveBeenCalledTimes(1)
    expect(search).toHaveBeenCalledWith('abc')
    expect(results.value).toEqual(['result:abc'])
    scope.stop()
  })
})

effectScope 让 watch 的生命周期独立于组件，测试结束 scope.stop() 即可清理，避免用例间串扰。

三、组件测试：只测「输入 → 输出」 ​

组件测试环境需要 jsdom 与 @vue/test-utils：

bash
pnpm add -D @vue/test-utils jsdom
ts
// tests/components/QuantityStepper.test.ts
// @vitest-environment jsdom
import { describe, expect, it } from 'vitest'
import { mount } from '@vue/test-utils'
import QuantityStepper from '@/components/QuantityStepper.vue'

describe('QuantityStepper', () => {
  it('点击加号触发 update:modelValue', async () => {
    const wrapper = mount(QuantityStepper, { props: { modelValue: 2, min: 1, max: 5 } })
    await wrapper.find('[data-test="plus"]').trigger('click')
    expect(wrapper.emitted('update:modelValue')?.[0]).toEqual([3])
  })

  it('达到上限后加号禁用', () => {
    const wrapper = mount(QuantityStepper, { props: { modelValue: 5, min: 1, max: 5 } })
    expect(wrapper.find('[data-test="plus"]').attributes('disabled')).toBeDefined()
  })
})

组件测试的三条边界：

  • 不用快照测整页：DOM 一改快照就红，维护成本高于收益。
  • 不测样式细节（颜色、像素），交给视觉回归工具。
  • 业务编排放用例层测，组件只测 props / emits / 插槽。

四、契约测试：前端与后端各退一步 ​

单元测试无法发现「后端把 sale_price 改成 salePrice」。契约测试守住接口形状，分两层：

4.1 本地契约快照（零依赖，随代码库走）：

ts
// tests/contract/goods.contract.test.ts
import { describe, expect, it } from 'vitest'
import fixture from './fixtures/goods-list.json'

interface GoodsDto {
  goods_id: string
  goods_name: string
  sale_price: number
  stock_num: number
}

function assertGoodsList(input: unknown): asserts input is GoodsDto[] {
  if (!Array.isArray(input)) throw new Error('响应必须是数组')
  for (const item of input) {
    if (typeof item.goods_id !== 'string') throw new Error('goods_id 必须是 string')
    if (typeof item.sale_price !== 'number') throw new Error('sale_price 必须是 number（分）')
    if (!Number.isInteger(item.sale_price) || item.sale_price < 0) {
      throw new Error('sale_price 必须是非负整数（分）')
    }
  }
}

describe('商品列表契约', () => {
  it('线上采样 fixture 仍符合当前 DTO', () => {
    expect(() => assertGoodsList(fixture)).not.toThrow()
  })
})

把线上真实响应（脱敏后）存成 fixture，后端一改字段，CI 立刻失败。

4.2 消费者驱动契约：前后端都接入 Pact 时，前端生成契约文件（pact.json），后端在 CI 中验证自己满足契约。适合跨团队、接口多的项目；小团队用 4.1 的快照即可。

五、覆盖率：先棘轮，再提升 ​

仓库的 vitest.config.ts 已配置覆盖率门禁：

ts
// vitest.config.ts 中 test.coverage 的配置片段（此处以独立常量展示）
export const coverage = {
  provider: 'v8',
  all: true,
  thresholds: { statements: 70, branches: 85, functions: 88, lines: 70 }
}
bash
pnpm test:coverage
# 低于阈值时 vitest 直接非零退出，CI 变红

当前是「只升不降」的棘轮基线：每次重构补测试后，把阈值往上调 2–3 个百分点并提交。不要一开始就设 80%——达不到的阈值会被团队直接关掉，反而失去约束。

六、验收标准 ​

bash
pnpm test
# 期望：store / composable 用例通过，无真实网络与定时器等待

pnpm test:coverage
# 期望：命中阈值，且失败路径用例占比 ≥ 30%

自查：把 API mock 去掉换成真实网络，测试应当立即失败——说明 mock 生效、边界清晰；如果仍然通过，说明测试没有真正覆盖副作用。
