4.2 进阶应用：企业级请求层 ​

一个合格的请求层要解决五件事：统一前缀、统一注入、统一解包、统一错误、统一重试。

请求内核：配置 + 拦截器链 ​

ts
// src/api/request.ts
import { useUserStore } from '@/stores/user'
import { env } from '@/utils/env'

export interface RequestConfig {
  url: string
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE'
  data?: Record<string, unknown>
  header?: Record<string, string>
  timeout?: number
  retry?: number          // 失败重试次数（仅幂等 GET 默认 1 次）
  skipAuth?: boolean      // 不注入 token
  skipErrorToast?: boolean
  signal?: { aborted: boolean } // 简化的取消标记
}

type RequestInterceptor = (config: RequestConfig) => RequestConfig
type ResponseInterceptor = (data: unknown, config: RequestConfig) => unknown

const requestInterceptors: RequestInterceptor[] = []
const responseInterceptors: ResponseInterceptor[] = []

export function onRequest(fn: RequestInterceptor) {
  requestInterceptors.push(fn)
}
export function onResponse(fn: ResponseInterceptor) {
  responseInterceptors.push(fn)
}

function sleep(ms: number) {
  return new Promise((r) => setTimeout(r, ms))
}

function rawRequest<T>(config: RequestConfig): Promise<T> {
  return new Promise((resolve, reject) => {
    const task = uni.request({
      url: config.url,
      method: config.method,
      data: config.data,
      header: config.header,
      timeout: config.timeout ?? 12000,
      success: (res) => {
        if (res.statusCode >= 200 && res.statusCode < 300) resolve(res.data as T)
        else reject(Object.assign(new Error(`HTTP ${res.statusCode}`), { statusCode: res.statusCode, data: res.data }))
      },
      fail: (err) => reject(new Error(err.errMsg))
    })
    // 取消标记：业务层可通过 AbortLike 控制
    if (config.signal) {
      const timer = setInterval(() => {
        if (config.signal!.aborted) {
          task.abort()
          clearInterval(timer)
        }
      }, 50)
      setTimeout(() => clearInterval(timer), config.timeout ?? 12000)
    }
  })
}

async function request<T>(config: RequestConfig): Promise<T> {
  // 1. 拼接 baseURL
  config.url = env.apiBase + config.url
  config.header = { 'Content-Type': 'application/json' }

  // 2. 请求拦截器链
  for (const fn of requestInterceptors) config = fn(config)

  const maxRetry = config.retry ?? (config.method === 'GET' || !config.method ? 1 : 0)
  let attempt = 0
  let lastError: unknown

  while (attempt <= maxRetry) {
    try {
      let data = await rawRequest<T>(config)
      // 3. 响应拦截器链
      for (const fn of responseInterceptors) data = fn(data, config) as T
      return data
    } catch (err) {
      lastError = err
      const status = (err as { statusCode?: number }).statusCode
      // 401 不重试；4xx 不重试；仅网络错误/5xx 指数退避
      if (status === 401) throw err
      if (status && status < 500) throw err
      if (attempt === maxRetry) break
      await sleep(300 * 2 ** attempt)
      attempt++
    }
  }
  throw lastError
}

export const http = {
  get<T>(url: string, config: Partial<RequestConfig> = {}) {
    return request<T>({ ...config, url, method: 'GET' })
  },
  post<T>(url: string, data?: Record<string, unknown>, config: Partial<RequestConfig> = {}) {
    return request<T>({ ...config, url, method: 'POST', data })
  }
}

请求内核的四个工程化能力 ​

拦截器解决「统一」，但真实项目还会遇到并发过高、同参重复请求、缓存过期、写操作重复提交、线上无法定位慢请求。这四件事都应该在内核层解决，而不是散落在页面里。

并发控制：信号量限流 ​

小程序和 App 的并发连接数有限，首屏同时发 20 个请求会互相排队，反而拉长整体耗时。用一个最小信号量把并发压到 6：

ts
// src/api/semaphore.ts
export function createSemaphore(limit: number) {
  let active = 0
  const queue: Array<() => void> = []

  return async function run<T>(task: () => Promise<T>): Promise<T> {
    if (active >= limit) {
      await new Promise<void>((resolve) => queue.push(resolve))
    }
    active++
    try {
      return await task()
    } finally {
      active--
      queue.shift()?.()
    }
  }
}
ts
// src/api/request.ts 末尾
const limit = createSemaphore(6)

export function requestLimited<T>(config: RequestConfig) {
  // 上传/下载等长任务不要走这里，避免占满名额
  return limit(() => request<T>(config))
}

高优先级请求（登录、支付查询）可以单独建一个 createSemaphore(2) 的高优队列，避免被列表请求堵住。

GET 去重与缓存：三个 Map 解决 80% 的重复流量 ​

同一页面多个组件同时请求同一份配置、快速切换 tab 重复拉列表，都是典型的重复流量。用「进行中 Promise + TTL 缓存」合并：

ts
// src/api/cache.ts
interface CacheEntry {
  data: unknown
  expiresAt: number
}

const inflight = new Map<string, Promise<unknown>>()
const cache = new Map<string, CacheEntry>()

function cacheKey(config: RequestConfig) {
  // 缓存键必须包含用户维度，避免 A 用户看到 B 用户的私有数据
  const uid = uni.getStorageSync('uid') || 'anon'
  return `${uid}|${config.method ?? 'GET'}|${config.url}|${JSON.stringify(config.data ?? {})}`
}

export async function requestDeduped<T>(
  config: RequestConfig,
  fetchFn: (c: RequestConfig) => Promise<T>,
  options: { ttl?: number; force?: boolean } = {}
): Promise<T> {
  const key = cacheKey(config)
  const hit = cache.get(key)
  if (!options.force && hit && hit.expiresAt > Date.now()) {
    return hit.data as T
  }

  const running = inflight.get(key)
  if (running) return running as Promise<T>

  const task = fetchFn(config)
    .then((data) => {
      cache.set(key, { data, expiresAt: Date.now() + (options.ttl ?? 30_000) })
      return data
    })
    .finally(() => inflight.delete(key))

  inflight.set(key, task)
  return task
}

/** 写操作成功后按前缀失效，例如 invalidate('/goods') */
export function invalidateCache(prefix: string) {
  for (const key of cache.keys()) {
    if (key.includes(prefix)) cache.delete(key)
  }
}

缓存策略必须写清楚，否则会变成「改了数据前端不更新」的 bug：

数据类型策略TTL失效方式
配置 / 字典 / 地区缓存优先，后台刷新5–30 分钟版本号变更或手动清缓存
商品列表 / 详情网络优先，失败回落缓存30–60 秒下单、改库存后按 /goods 前缀失效
用户资料 / 订单不缓存或极短 TTL0–10 秒每次进入页面强制刷新
支付结果绝不缓存—只能实时查询服务端

POST 幂等键：客户端生成，服务端唯一约束 ​

UI 层 submitting 锁只能防同一页面连点；网络超时重试、用户杀进程后重进、多端并发提交仍会产生重复订单。正确做法是给每次「业务意图」生成一个幂等键，重试时复用同一个键：

ts
// src/api/idempotency.ts
export function createIdempotencyKey(scope: string) {
  const rand = Math.random().toString(36).slice(2, 10)
  return `${scope}-${Date.now()}-${rand}`
}
ts
// 页面提交：key 在进入确认页时生成一次，重试沿用
const idempotencyKey = createIdempotencyKey('order')

async function submit() {
  await http.post('/orders', payload, {
    header: { 'Idempotency-Key': idempotencyKey }
  })
}

服务端用唯一索引兜底（PostgreSQL 示例）：

sql
CREATE UNIQUE INDEX uk_idempotency_key ON idempotency (key);

-- 幂等写入：插入成功才执行业务；冲突则返回已有结果
INSERT INTO idempotency (key, user_id, request_hash, response_json, created_at)
VALUES ($1, $2, $3, NULL, now())
ON CONFLICT (key) DO NOTHING;

-- 影响行数 = 0：说明是重放，直接读取已有 response_json 返回

约束：同一个幂等键只能对应同一份请求体。服务端应校验 request_hash，键相同但参数不同直接返回 409，避免「用旧键提交了新订单」。

请求级埋点：每次请求都要能回答「慢在哪」 ​

只在前端 console.time 无法定位线上问题。内核层为每个请求生成 requestId，记录耗时、重试、是否命中缓存，批量上报：

ts
// src/api/telemetry.ts
export interface RequestSample {
  requestId: string
  url: string
  method: string
  status: number | 'network-error'
  durationMs: number
  retry: number
  cached: boolean
  at: number
}

const buffer: RequestSample[] = []

export function recordRequest(sample: RequestSample) {
  buffer.push(sample)
  if (buffer.length >= 50) flushRequests()
}

function flushRequests() {
  if (!buffer.length) return
  const batch = buffer.splice(0, buffer.length)
  // #ifdef H5
  navigator.sendBeacon?.('/log/request', JSON.stringify(batch))
  // #endif
  // #ifndef H5
  uni.request({ url: '/log/request', method: 'POST', data: batch })
  // #endif
}

在 request() 里补充采样（示意，接在 try/finally 上）：

ts
const requestId = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`
const startedAt = Date.now()

try {
  const data = await rawRequest<T>(config)
  recordRequest({
    requestId,
    url: config.url,
    method: config.method ?? 'GET',
    status: 200,
    durationMs: Date.now() - startedAt,
    retry: attempt,
    cached: false,
    at: startedAt
  })
  return data
} catch (err) {
  // 失败也要采样，网络错误用 'network-error' 标记
  throw err
}

生产环境要按 requestId 同时打通前端埋点与服务端日志（响应头回传 X-Request-Id），才能做端到端排查。完整方案（批量、采样、离线补传、脱敏与 trace 串联）见 24.1 Telemetry SDK 设计。

拦截器：注入 token、解包、错误归一 ​

ts
// src/api/interceptors.ts
import { onRequest, onResponse } from './request'
import { useUserStore } from '@/stores/user'
import type { ApiResponse } from './types'

// 静默刷新状态（防并发刷新）
let refreshing: Promise<string> | null = null

onRequest((config) => {
  // 演示用参数控制；真实项目通过单独的 skipAuth 透传
  if (!(config as Record<string, unknown>).skipAuth) {
    const token = uni.getStorageSync('token')
    if (token) config.header!.Authorization = `Bearer ${token}`
  }
  return config
})

onResponse((data) => {
  const body = data as ApiResponse
  if (body.code === 0) return body.data
  if (body.code === 401) {
    throw Object.assign(new Error('登录已过期'), { statusCode: 401 })
  }
  uni.showToast({ title: body.message || '请求失败', icon: 'none' })
  throw new Error(body.message)
})

export async function refreshToken(): Promise<string> {
  if (refreshing) return refreshing
  const userStore = useUserStore()
  refreshing = http
    .post<{ token: string }>('/auth/refresh', { refreshToken: userStore.refreshToken })
    .then((res) => {
      uni.setStorageSync('token', res.token)
      return res.token
    })
    .finally(() => {
      refreshing = null
    })
  return refreshing
}

401 刷新后重放原请求，需要在 request 的 catch 中补一段（注意只重放一次）：

ts
// request() 内 catch 分支
if (status === 401 && !(config as { __retried?: boolean }).__retried) {
  await refreshToken()
  ;(config as { __retried?: boolean }).__retried = true
  return request<T>(config)
}

防重复提交 ​

写操作在 UI 层加 loading 锁即可解决 80% 的重复提交：

ts
const submitting = ref(false)
async function submit() {
  if (submitting.value) return
  submitting.value = true
  try {
    await orderApi.create(payload)
    uni.showToast({ title: '下单成功' })
  } finally {
    submitting.value = false
  }
}

更彻底的方案是请求层对相同 url+body 的 POST 做短时去重，可按需加入。

常见问题 ​

Q：拦截器里怎么拿 Pinia store？ 在拦截器函数内部调用 useUserStore()（请求发生时 Pinia 已安装），不要在模块顶层调用。

Q：文件上传走同一套封装吗？uni.uploadFile 是独立 API，建议在 request 目录平行实现 upload.ts，复用同样的 token 注入与错误提示，但不要硬塞进 uni.request 封装。

Q：重试要注意什么？ 只对幂等请求（GET/PUT/DELETE）自动重试；POST 自动重试可能造成重复下单。指数退避 + 最大次数（1–2 次足够）是性价比最高的配置。
